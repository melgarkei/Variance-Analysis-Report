function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var result = []; // 1. EXTRACT DATA FOR VARIANCE REPORT var varianceSheet = ss.getSheetByName("Payment Proposals 2026") || ss.getSheets()[0]; if (varianceSheet) { var vData = varianceSheet.getDataRange().getValues(); for (var i = 1; i < vData.length; i++) { var row = vData[i]; var week = row[0]; // Col A: Week var month = row[1]; // Col B: Month var entity = row[5]; // Col F: Entity var currency = row[7]; // Col H: Currency var localAmt = row[8] || row[9]; // Col I/J: Local Amount var sarAmt = row[12]; // Col M: SAR Amount var category = row[13]; // Col N: Category var type = row[14] ? String(row[14]).toLowerCase().trim() : 'actual'; if (week || month) { result.push({ type: type, week: String(week || ""), month: String(month || ""), entity: String(entity || ""), currency: String(currency || "SAR"), localAmt: Number(localAmt) || 0, sarAmt: Number(sarAmt) || 0, category: String(category || "") }); } } } // 2. EXTRACT PROJECTIONS FOR PHASE 1 COMPARISON var projSheet = ss.getSheetByName("Copy of Monthly Projection 2026"); if (projSheet) { var pData = projSheet.getDataRange().getValues(); for (var j = 1; j < pData.length; j++) { var pRow = pData[j]; var pWeek = pRow[0]; // Col A var pMonth = pRow[1]; // Col B var pEntity = pRow[5]; // Col F var pCurrency = pRow[7]; // Col H var pLocalAmt = pRow[9]; // Col J var pSarAmt = pRow[12]; // Col M var pCategory = pRow[13]; // Col N if (pWeek || pMonth) { result.push({ type: 'projection', week: String(pWeek || ""), month: String(pMonth || ""), entity: String(pEntity || ""), currency: String(pCurrency || "SAR"), localAmt: Number(pLocalAmt) || 0, sarAmt: Number(pSarAmt) || 0, category: String(pCategory || "") }); } } } var callback = e ? e.parameter.callback : null; var jsonOutput = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + '(' + jsonOutput + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonOutput) .setMimeType(ContentService.MimeType.JSON); } }
