function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); // Try reading from 'Copy of Monthly Projection 2026', fallback to active sheet var sheet = ss.getSheetByName("Copy of Monthly Projection 2026") || ss.getActiveSheet(); var sheetName = sheet.getName(); var data = sheet.getDataRange().getValues(); var result = []; // Set category column index depending on the tab name // Column L = Index 11 | Column Q = Index 16 var categoryIdx = (sheetName === "Copy of Monthly Projection 2026") ? 11 : 16; // Loop through rows (skipping row 0 header) for (var i = 1; i < data.length; i++) { var row = data[i]; var weekVal = String(row[0] || "").trim(); var entity = String(row[5] || "Unknown").trim(); var currency = String(row[8] || "SAR").trim(); var projLocal = Number(row[9]) || 0; var actLocal = Number(row[10]) || 0; var sarAmt = Number(row[11]) || 0; // Dynamically pull category based on tab mapping var category = String(row[categoryIdx] || "Uncategorized").trim(); if (weekVal !== "") { result.push({ week: weekVal, entity: entity || "Unknown", category: category || "Uncategorized", currency: currency || "SAR", projLocal: projLocal, actLocal: actLocal, sarAmt: sarAmt }); } } // Support JSONP for cross-domain dashboard fetching var callback = e && e.parameter ? e.parameter.callback : null; var jsonString = JSON.stringify(result); if (callback) { return ContentService.createTextOutput(callback + "(" + jsonString + ")") .setMimeType(ContentService.MimeType.JAVASCRIPT); } else { return ContentService.createTextOutput(jsonString) .setMimeType(ContentService.MimeType.JSON); } }
