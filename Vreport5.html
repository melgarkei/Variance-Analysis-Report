function updateDashboard() { if (!rawData.length) return; const ALLOWED_CATEGORIES = [ "Trade - Amex", "Trade - Vcc", "Trade - VCC", "Customer Refunds", "Trade - Non Air", "Non Trade - Other", "Trade Air - BSP", "Non Trade - Payroll", "Trade Air -GSA", "Trade Air - GSA", "Trade Air - LCC" ]; const selWeek = document.getElementById('weekSelect').value; const selEntity = document.getElementById('entitySelect').value; const selCat = document.getElementById('categorySelect').value; const currMode = document.getElementById('currencyMode').value; const groupBy = document.getElementById('groupSelect').value; const filtered = rawData.filter(d => { const matchWeek = (selWeek === "All" || d.week === selWeek); const matchEntity = (selEntity === "All" || d.entity === selEntity); const matchCat = (selCat === "All" || d.category === selCat); // Only allow valid categories if grouping by Category const isValidCat = (groupBy === "Category") ? ALLOWED_CATEGORIES.includes(d.category) : true; return matchWeek && matchEntity && matchCat && isValidCat; }); const aggregated = {}; filtered.forEach(item => { const key = groupBy === "Category" ? item.category : item.entity; if (!aggregated[key]) { aggregated[key] = { projected: 0, actual: 0, curr: currMode === "SAR" ? "SAR" : (item.currency || "Local") }; } const amount = (currMode === "SAR") ? (item.sarAmt || 0) : (item.localAmt || 0); if (item.type === 'projection') { aggregated[key].projected += amount; } else if (item.type === 'actual') { aggregated[key].actual += amount; } }); const labels = Object.keys(aggregated); const projectedData = labels.map(l => aggregated[l].projected); const actualData = labels.map(l => aggregated[l].actual); renderChart(labels, projectedData, actualData, currMode); renderTable(aggregated, groupBy, currMode); }
